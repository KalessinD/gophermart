// Интерфейс в сервисе
type UnitOfWork interface {
    OrderRepository() OrderRepository
    UserRepository() UserRepository
    Commit() error
}

// Логика в сервисе
func (s *Service) ProcessOrder(order *models.Order) error {
    return s.uow.Execute(ctx, func(uow UnitOfWork) error {
        if err := uow.OrderRepository().Update(order); err != nil {
            return err
        }
        
        // Бизнес-логика тут
        if order.Status == models.OrderProcessedStatus && order.Accrual > 0 {
            if err := uow.UserRepository().AddBalance(order.UserID, order.Accrual); err != nil {
                return err
            }
        }
        return nil
    })
}
